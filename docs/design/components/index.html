Filename and language share a header with the copy action. Python and JavaScript are highlighted in the browser; code stays readable without JavaScript or when highlighting is unavailable.

relay.pyPython
import json


def encode_message(text: str) -> bytes:
    # Keep the message boundary explicit.
    message = {"type": "message", "text": text, "version": 1}
    return json.dumps(message).encode("utf-8")


payload = encode_message("Hello, NodalWire 👋")
print(payload)
message.jsJavaScript
function encodeMessage(text) {
  // Preserve literal text, including <tags> & whitespace.
  const message = { type: "message", text, version: 1 };
  return new TextEncoder().encode(JSON.stringify(message));
}

const payload = encodeMessage("Hello, NodalWire 👋");
console.log(payload);
